% Minimal recreation of the clamped-frame demo (satopdf only)
\documentclass[tikz]{standalone}
\usepackage{satopdf}
\usetikzlibrary{calc}

\tikzset{stpFrame/.pic={
  \stpPoint{A}{0}{3}\stpPoint{B}{3}{3}\stpPoint{C}{6}{3}\stpPoint{D}{9}{3}
  \stpPoint{E}{3}{0}\stpPoint{F}{6}{0}
  % 刚架：beam 固接（无铰圆）；墙式固支
  \foreach \a/\b in {A/B,B/C,C/D,E/B,F/C}{\stpMember{beam}{\a}{\b}}
  % 单梁端全固支墙：type=wall；angle = 外侧法向
  \stpSupport{A}[type=wall, angle=-90]
  \stpSupport{D}[type=wall, angle=90]
  \stpSupport{E}[type=wall]
  \stpSupport{F}[type=wall]
  \foreach \n/\a in {A/below right,B/below right,C/below right,
                     D/below left,E/above right,F/above right}
    {\stpNotation{node}{\n}{\n}[\a]}
}}
